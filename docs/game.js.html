/**
 * @module game
 * @description
 * Hiking Survival — core game module.
 *
 * All exported functions are **pure**: given the same arguments they always
 * return the same value and produce no observable side‑effects.  The UI layer
 * (app.js) must not implement any game logic; it should only call these
 * functions, render the returned state, and display the returned event log.
 */

// ── JSDoc type definitions ────────────────────────────────────────────────────

/**
 * @typedef {'playing'|'won'|'over'} GameStatus
 * The current outcome of the expedition.
 * - `'playing'` — the run is in progress.
 * - `'won'`     — the player reached the summit.
 * - `'over'`    — at least one survival stat reached zero.
 */

/**
 * @typedef {Object} Gear
 * @property {boolean} crampons - Whether ice‑grip crampons are equipped.
 */

/**
 * @typedef {Object} Player
 * @property {number} position - Index into the trail array (0 = Base Camp).
 * @property {number} stamina  - Physical energy (0–100). Reaching 0 = death.
 * @property {number} hunger   - Nutritional reserves (0–100). Reaching 0 = death.
 * @property {number} warmth   - Body heat (0–100). Reaching 0 = death.
 * @property {Gear}   gear     - Permanently equipped gear items.
 */

/**
 * @typedef {Object} InventoryItem
 * @property {string}  id      - Unique identifier for the item type.
 * @property {string}  name    - Display name (may include a leading emoji).
 * @property {string}  type    - Effect category: 'stamina'|'hunger'|'warmth'|'gear'|'camp'.
 * @property {number}  [value] - Stat restoration amount (omitted for gear/camp items).
 * @property {string}  [gear]  - Player gear key to set to true (gear items only).
 * @property {string}  [image] - Path to a custom icon image.
 */

/**
 * @typedef {Object} TrailNode
 * @property {number} id    - Sequential index, matches its position in the trail array.
 * @property {number} day   - Expedition day this node belongs to (1–5).
 * @property {number} x     - Map pin x‑position as a percentage of the map width.
 * @property {number} y     - Map pin y‑position as a percentage of the map height.
 * @property {string} name  - Human‑readable location name.
 * @property {string} type  - Terrain category: 'start'|'normal'|'hard'|'danger'|'camp'|'summit'.
 * @property {string} image - URL or local path for the scene background image.
 */

/**
 * @typedef {Object} GameState
 * @property {Player}          player    - Current player stats and position.
 * @property {InventoryItem[]} inventory - Items remaining in the backpack.
 * @property {TrailNode[]}     trail     - Full immutable trail definition.
 * @property {GameStatus}      status    - Current game outcome.
 */

/**
 * @typedef {Object} ActionResult
 * @property {GameState} state  - New game state produced by the action.
 * @property {string[]}  events - Ordered log messages generated by the action.
 */

/**
 * Return a fresh deep copy of the initial game state.
 * Use this to start a new game; mutating the returned object never affects
 * `initialState` or any other copy.
 * @returns {GameState}
 */
export function getNewGame() {
    return JSON.parse(JSON.stringify(initialState));
}

// ── Pure utility helpers ───────────────────────────────────────────────────────

/**
 * Clamp a stat value to the valid range [0, 100].
 * @param {number} value - Raw stat value after applying a delta.
 * @returns {number} Value clamped to [0, 100].
 */
export function clampStat(value) {
    return Math.max(0, Math.min(100, value));
}

/**
 * Return whether a trail node has icy terrain based on its name.
 * Icy terrain inflicts an extra warmth penalty when traversed.
 * @param {TrailNode} cell - The trail node to inspect.
 * @returns {boolean}
 */
export function isIcyCell(cell) {
    const n = cell.name.toLowerCase();
    return n.includes("ice") || n.includes("icy") ||
           n.includes("snow") || n.includes("frozen");
}

/**
 * Format a set of stat deltas into a compact human‑readable cost string,
 * e.g. `"−7 Stam  −4 Food  +5 Warm"`.  Components whose delta is 0 are omitted.
 * @param {number} s - Stamina delta.
 * @param {number} h - Hunger delta.
 * @param {number} w - Warmth delta.
 * @returns {string} Formatted cost line, or an empty string if all deltas are zero.
 */
export function formatCostLine(s, h, w) {
    const fmt = (val, label) => {
        if (val === 0) return null;
        return (val > 0 ? "+" : "−") + Math.abs(val) + " " + label;
    };
    return [fmt(s, "Stam"), fmt(h, "Food"), fmt(w, "Warm")]
        .filter(Boolean)
        .join("  ");
}

// ── State query functions ─────────────────────────────────────────────────────

/**
 * Return the trail node the player is currently standing on.
 * @param {GameState} state
 * @returns {TrailNode}
 */
export function getCurrentCell(state) {
    return state.trail[state.player.position];
}

/**
 * Return the next trail node ahead of the player, or `null` if already at the end.
 * @param {GameState} state
 * @returns {TrailNode|null}
 */
export function getNextCell(state) {
    const next = state.player.position + 1;
    return next < state.trail.length ? state.trail[next] : null;
}

/**
 * Return whether the player is currently standing on a camp node.
 * Camp & Rest may only be performed at camp nodes.
 * @param {GameState} state
 * @returns {boolean}
 */
export function isAtCamp(state) {
    return getCurrentCell(state).type === "camp";
}

/**
 * Return whether the player has already rested at the current camp node.
 * Each camp may only be used for Camp & Rest once per expedition.
 * @param {GameState} state
 * @returns {boolean}
 */
export function hasRestedHere(state) {
    return state.restedAtCamps.includes(getCurrentCell(state).id);
}

/**
 * Derive the current game status purely from player stats and position.
 * - `'over'`    if any survival stat (stamina, hunger, warmth) has reached 0.
 * - `'won'`     if the player has reached the final trail node (the summit).
 * - `'playing'` otherwise.
 * @param {GameState} state
 * @returns {GameStatus}
 */
export function checkStatus(state) {
    const { stamina, hunger, warmth } = state.player;
    if (stamina <= 0 || hunger <= 0 || warmth <= 0) return "over";
    if (state.player.position === state.trail.length - 1) return "won";
    return "playing";
}

// ── Internal helpers (not exported) ──────────────────────────────────────────

/**
 * Return a capitalised display name for a stat type.
 * @param {'stamina'|'hunger'|'warmth'} type
 * @returns {string}
 */
function getStatName(type) {
    if (type === "stamina") return "Stamina";
    if (type === "hunger")  return "Hunger";
    return "Warmth";
}

/**
 * Apply terrain and gear effects for a cell, returning updated player stats
 * and the log events they generate.
 * @param {Player}    player
 * @param {TrailNode} cell
 * @returns {{ player: Player, events: string[] }}
 */
function applyTerrainEffect(player, cell) {
    const effect = terrainEffects[cell.type];
    if (!effect) return { player, events: [] };

    const events = [];
    let staminaDelta = effect.stamina;
    let warmthDelta  = effect.warmth;

    if (isIcyCell(cell)) {
        warmthDelta -= 5;
        events.push("🧊 The freezing terrain drains your body heat even faster.");
    }
    if (player.gear.crampons && isIcyCell(cell)) {
        staminaDelta += 5;
        events.push("🥾 Your crampons grip the ice, saving you some stamina.");
    }

    const label    = terrainPenaltyLabels[cell.type] || "Terrain";
    const costLine = formatCostLine(staminaDelta, effect.hunger, warmthDelta);
    if (costLine) events.push("  ↳ " + label + ": " + costLine);
    events.push(effect.message);

    return {
        player: {
            ...player,
            stamina: clampStat(player.stamina + staminaDelta),
            hunger:  clampStat(player.hunger  + effect.hunger),
            warmth:  clampStat(player.warmth  + warmthDelta)
        },
        events
    };
}

// ── Action functions ──────────────────────────────────────────────────────────

/**
 * Advance the player one step along the trail, applying the base movement
 * cost, terrain effects, and a deterministic random event.
 *
 * Has no effect and returns the original state when the game is not in the
 * `'playing'` status, or when the player is already at the final node.
 *
 * Base cost per move: −7 Stamina · −4 Hunger · −4 Warmth.
 * Additional costs depend on the terrain type and any active random event.
 *
 * @param {GameState} state - Current game state (not mutated).
 * @returns {ActionResult} New state and log events.
 */
export function moveForward(state) {
    if (state.status !== "playing") return { state, events: [] };

    const nextPos = state.player.position + 1;
    if (nextPos >= state.trail.length) return { state, events: [] };

    const cell   = state.trail[nextPos];
    const events = [];

    // 1 — Base movement cost (waived when the destination is a safe camp node)
    let player;
    if (cell.type === "camp") {
        player = { ...state.player, position: nextPos };
        events.push("⛺ The camp is a safe haven — no movement cost to arrive.");
        events.push("⛺ You reach [" + cell.name + "] and drop your pack with relief.");
    } else {
        events.push("🚶 Base move cost:  −7 Stam  −4 Food  −4 Warm");
        player = {
            ...state.player,
            position: nextPos,
            stamina: clampStat(state.player.stamina - 7),
            hunger:  clampStat(state.player.hunger  - 4),
            warmth:  clampStat(state.player.warmth  - 4)
        };
        events.push("🚶‍♂️ Pushing through the biting wind, you reach [" + cell.name + "].");
    }

    // 2 — Terrain effect
    const terrainResult = applyTerrainEffect(player, cell);
    player = terrainResult.player;
    events.push(...terrainResult.events);

    // 3 — Deterministic surprise event (camp and summit nodes are exempt)
    if (cell.type !== "camp" && cell.type !== "summit") {
        const ev     = surpriseEvents[cell.id % surpriseEvents.length];
        const evCost = formatCostLine(ev.stamina, ev.hunger, ev.warmth);
        if (evCost) events.push("  ↳ Event penalty: " + evCost);
        player = {
            ...player,
            stamina: clampStat(player.stamina + ev.stamina),
            hunger:  clampStat(player.hunger  + ev.hunger),
            warmth:  clampStat(player.warmth  + ev.warmth)
        };
        events.push(ev.message);
    }

    const newState = { ...state, player };
    return { state: { ...newState, status: checkStatus(newState) }, events };
}

/**
 * Rest at the current location to recover Stamina and Warmth at the cost of
 * increased Hunger.
 *
 * Recovery: +30 Stamina · +20 Warmth · −15 Hunger.
 * Stats cannot exceed 100 or drop below 0.
 *
 * Has no effect when the game is not in the `'playing'` status, or when the
 * player is not currently standing on a camp node.
 *
 * @param {GameState} state - Current game state (not mutated).
 * @returns {ActionResult} New state and log events.
 */
export function campAndRest(state) {
    if (state.status !== "playing") return { state, events: [] };
    if (!isAtCamp(state)) {
        return { state, events: ["⛺ You can only rest at a designated camp node."] };
    }
    if (hasRestedHere(state)) {
        return { state, events: ["⛺ You've already rested at this camp. Move on to the next stage."] };
    }

    const campId = getCurrentCell(state).id;
    const events = ["⛺ Camp & Rest:  +30 Stam  +20 Warm  −15 Food"];

    const player = {
        ...state.player,
        stamina: clampStat(state.player.stamina + 30),
        warmth:  clampStat(state.player.warmth  + 20),
        hunger:  clampStat(state.player.hunger  - 15)
    };

    events.push("⛺ You set up camp and rest. The campfire restores your warmth and stamina, but your stomach is completely empty.");

    const newState = { ...state, player, restedAtCamps: [...state.restedAtCamps, campId] };
    return { state: { ...newState, status: checkStatus(newState) }, events };
}

/**
 * Use an item from the player's backpack by its index in the inventory array.
 * The item is permanently removed from the inventory after use.
 *
 * - **Stamina / hunger / warmth items** restore the corresponding stat immediately.
 * - **Gear items** (e.g. Crampons) are equipped permanently and reduce
 *   terrain penalties for the rest of the run.
 * - **Camp items** (e.g. Bivouac Tent) can only be used during Camp & Rest;
 *   calling this function with a camp item returns an explanatory event and
 *   leaves the state unchanged.
 *
 * Has no effect when the game is not in the `'playing'` status, or when the
 * index is out of range.
 *
 * @param {GameState} state - Current game state (not mutated).
 * @param {number}    index - Zero‑based index into `state.inventory`.
 * @returns {ActionResult} New state and log events.
 */
export function useItem(state, index) {
    if (state.status !== "playing") return { state, events: [] };
    if (index < 0 || index >= state.inventory.length) return { state, events: [] };

    const item   = state.inventory[index];
    const events = [];

    // Camp items require a Camp & Rest action — do not consume
    if (item.type === "camp") {
        events.push("⛺ Tents must be used during 'Camp & Rest'. Keep it in your pack for now.");
        return { state, events };
    }

    // Remove item from inventory (pure: build new array without this index)
    const newInventory = state.inventory.filter((_, i) => i !== index);

    // Gear items — equip and do not restore any stat
    if (item.type === "gear") {
        const player = {
            ...state.player,
            gear: { ...state.player.gear, [item.gear]: true }
        };
        events.push("🥾 You equipped [" + item.name + "]. Stamina drain on icy terrain will be reduced.");
        return { state: { ...state, player, inventory: newInventory }, events };
    }

    // Stat‑restoring items
    const player = {
        ...state.player,
        [item.type]: clampStat(state.player[item.type] + item.value)
    };
    events.push("🎒 You consumed [" + item.name + "], restoring " + item.value + " " + getStatName(item.type) + ".");

    const newState = { ...state, player, inventory: newInventory };
    return { state: { ...newState, status: checkStatus(newState) }, events };
}

/**
 * Apply one tick of the danger-zone passive drain: −1 Stamina · −1 Warmth.
 * Called every second while the player is standing on a danger node.
 *
 * Has no effect when the game is not in the `'playing'` status, or when the
 * current node is not of type `'danger'`.
 *
 * @param {GameState} state - Current game state (not mutated).
 * @returns {ActionResult} New state and an empty events array (drain is silent in the log).
 */
export function dangerDrain(state) {
    if (state.status !== "playing") return { state, events: [] };
    if (getCurrentCell(state).type !== "danger") return { state, events: [] };

    const player = {
        ...state.player,
        stamina: clampStat(state.player.stamina - 1),
        warmth:  clampStat(state.player.warmth  - 1)
    };

    const newState = { ...state, player };
    return { state: { ...newState, status: checkStatus(newState) }, events: [] };
}
